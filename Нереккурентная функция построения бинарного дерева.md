```python
from typing import Callable, Any


def gen_bin_tree(height: int = 3, root: int = 13,
                 l_b: Callable[[int], int] = lambda x: x + 1,
                 r_b: Callable[[int], int] = lambda y: y - 1) -> dict[int, list[Any]]:
    """Функция, которая возвращает бинарное дерево по заданным параметрам.

    :param int height: высота дерева
    :param int root: корень дерева
    :param function l_b: функция для создания левой ветки
    :param function r_b: функция для создания правой ветки
    :return: словарь, представляющий собой бинарное дерево
    :raises ValueError: если высота отрицательная

    > gen_bin_tree(2, 13)
    {13: [{14: [{15: []}, {13: []}]}, {12: [{13: []}, {11: []}]}]}
    """
    if height < 0:
        raise ValueError("Высота дерева не может быть отрицательным числом.")
    if height == 0:
        return {root: []}

    roots = [[root]]

    for leaf in range(height):
        r = roots[-1]
        leaves = list(map(lambda root_value: [l_b(root_value), r_b(root_value)], r))

        flat = []
        for s in leaves:
            for item in s:
                flat.append(item)

        roots.append(flat)

    tree_lvl = {i: [] for i in range(len(roots[-1]))}

    for depth in range(len(roots) - 2, -1, -1):
        new_lvl: dict[int, list[Any]] = {}
        for i, n in enumerate(roots[depth]):
            left_leaf = l_b(n)
            right_leaf = r_b(n)
            new_lvl[i] = [
                {left_leaf: tree_lvl[2 * i]},
                {right_leaf: tree_lvl[2 * i + 1]},
            ]
        tree_lvl = new_lvl

    return {root: tree_lvl[0]}
```
Она строит дерево нерекурсивным способом: сначала подготавливаются все "листья" дерева, а после они снизу вверх собираются в единое дерево.
Стоит отметить, что важно обратить внимание на то чтобы к веточкам обращались по индексу, а не по значению. Иначе, при существовании двух одинаковых листьев произошла бы ошибка построения дерева.

```python
import unittest
from binary_tree import gen_bin_tree


class TestGenBinTree(unittest.TestCase):

    def test_height_one(self):
        """Высота 1 - у корня два пустых поддерева."""
        result = gen_bin_tree(1, 5, lambda x: x + 5, lambda x: x - 5)
        self.assertEqual(result, {5: [{10: []}, {0: []}]})

    def test_height_two(self):
        """Высота 2 - у корня два поддерева, каждое из которых содержит по два пустых поддерева."""
        result = gen_bin_tree(2, 5, lambda x: x + 5, lambda x: x - 5)
        self.assertEqual(result, {5: [{10: [{15: []}, {5: []}]}, {0: [{5: []}, {-5: []}]}]})

    def test_negative_height(self):
        """Отрицательная высота - исключение ValueError."""
        with self.assertRaises(ValueError):
            gen_bin_tree(-1, 10, lambda x: x + 1, lambda x: x - 1)


if __name__ == '__main__':
    unittest.main()
```
